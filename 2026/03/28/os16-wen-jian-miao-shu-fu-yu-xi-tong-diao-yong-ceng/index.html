第16节：文件描述符与系统调用层

学习目标

  • 理解”一切皆文件”抽象的统一接口设计
  • 掌握文件描述符表（进程级）与全局文件表（内核级）的两层结构
  • 深入分析 file.c 中 filealloc/filedup/fileclose/fileread/filewrite 的实现
  • 理解管道（pipe）环形缓冲区与 sleep/wakeup 的配合
  • 掌握 sysfile.c 中主要系统调用的参数验证与执行流程

预计用时：3～4 小时
前置要求：第14节（日志层）、第15节（inode层）、第12节（锁机制）

一、”一切皆文件” 的统一抽象

Unix 最优雅的设计之一：所有 I/O 资源共享同一套接口——read/write/close。

用户空间                     内核空间
────────────────────────────────────────────────────────
  fd=3            ──→  proc->ofile[3]  ──→  struct file
                                              │ type=FD_PIPE    ──→  struct pipe (环形缓冲区)
                                              │ type=FD_INODE   ──→  struct inode (磁盘文件)
                                              │ type=FD_DEVICE  ──→  devsw[major].read/write
                                              └ type=FD_NONE    (未使用)

1.1 两层引用计数

层次一：进程文件描述符表

// kernel/proc.h
struct proc {
    struct file *ofile[NOFILE];   // 每个进程最多 16 个打开文件
    struct inode *cwd;            // 当前工作目录
    ...
};

层次二：全局文件表

// kernel/file.c
struct {
    struct spinlock lock;
    struct file file[NFILE];      // 系统全局最多 100 个文件对象
} ftable;

1.2 struct file 结构体

// kernel/file.h
struct file {
  enum { FD_NONE, FD_PIPE, FD_INODE, FD_DEVICE } type;
  int ref;         // 引用计数（有多少个 fd 指向此对象）
  char readable;   // 可读标志
  char writable;   // 可写标志
  struct pipe *pipe;  // type=FD_PIPE 时有效
  struct inode *ip;   // type=FD_INODE / FD_DEVICE 时有效
  uint off;           // 文件读写偏移量（仅 FD_INODE）
  short major;        // 设备主设备号（仅 FD_DEVICE）
};

关键常量（kernel/param.h）：

常量 值 含义
NOFILE 16 每进程最多打开文件数
NFILE 100 全局文件对象总数
NDEV 10 最大主设备号
MAXPATH 128 路径名最大长度
MAXARG 32 exec 最大参数数量

open flags（kernel/fcntl.h）：

常量 值 含义
O_RDONLY 0x000 只读
O_WRONLY 0x001 只写
O_RDWR 0x002 读写
O_CREATE 0x200 不存在则创建
O_TRUNC 0x400 截断为零

二、设备抽象：devsw 函数表

xv6 通过 devsw[] 函数指针表支持字符设备多态：

// kernel/file.h
struct devsw {
  int (*read)(int, uint64, int);   // 读函数指针
  int (*write)(int, uint64, int);  // 写函数指针
};

extern struct devsw devsw[];

#define CONSOLE 1   // console 的主设备号为 1

控制台驱动在 consoleinit() 中注册：

// kernel/console.c（第17节详细分析）
devsw[CONSOLE].read  = consoleread;
devsw[CONSOLE].write = consolewrite;

当对设备文件调用 fileread() 时，会通过 devsw[f->major].read 分发到对应驱动——这就是”一切皆文件”实现多态的核心机制。

三、file.c 源码精读

3.1 filealloc — 分配全局文件对象

struct file*
filealloc(void)
{
  struct file *f;

  acquire(&ftable.lock);          // 加全局锁，保护文件表
  for(f = ftable.file; f < ftable.file + NFILE; f++){
    if(f->ref == 0){              // 找到 ref==0 的空闲槽位
      f->ref = 1;                 // 初始引用计数为 1
      release(&ftable.lock);
      return f;
    }
  }
  release(&ftable.lock);
  return 0;                       // 全部占满，返回 NULL
}

注意：filealloc 只分配 struct file，不初始化 type/readable/writable/ip 等字段，调用者负责填写。

3.2 filedup — 增加引用计数

struct file*
filedup(struct file *f)
{
  acquire(&ftable.lock);
  if(f->ref < 1)
    panic("filedup");
  f->ref++;               // 原子性地递增引用计数
  release(&ftable.lock);
  return f;
}

dup() 系统调用和 fork() 都会调用 filedup，确保多个 fd 共享同一个 struct file 时不会提前释放。

3.3 fileclose — 释放引用计数

void
fileclose(struct file *f)
{
  struct file ff;

  acquire(&ftable.lock);
  if(f->ref < 1)
    panic("fileclose");
  if(--f->ref > 0){          // 还有其他引用者，仅减计数
    release(&ftable.lock);
    return;
  }
  // ref 降到 0，才真正关闭
  ff = *f;                   // 拷贝一份，用于锁外清理
  f->ref = 0;
  f->type = FD_NONE;         // 标记为空闲
  release(&ftable.lock);

  if(ff.type == FD_PIPE){
    pipeclose(ff.pipe, ff.writable);         // 关闭管道端
  } else if(ff.type == FD_INODE || ff.type == FD_DEVICE){
    begin_op();
    iput(ff.ip);             // 释放 inode 引用（可能触发磁盘 inode 释放）
    end_op();
  }
}

设计模式：先在锁内完成”标记释放”，再在锁外执行耗时操作（如 iput），避免持锁期间死锁。

3.4 fileread — 统一读接口

int
fileread(struct file *f, uint64 addr, int n)
{
  int r = 0;

  if(f->readable == 0)       // 权限检查
    return -1;

  if(f->type == FD_PIPE){
    r = piperead(f->pipe, addr, n);           // 管道读
  } else if(f->type == FD_DEVICE){
    if(f->major < 0 || f->major >= NDEV || !devsw[f->major].read)
      return -1;
    r = devsw[f->major].read(1, addr, n);     // 设备读（通过函数指针）
  } else if(f->type == FD_INODE){
    ilock(f->ip);
    if((r = readi(f->ip, 1, addr, f->off, n)) > 0)
      f->off += r;            // 更新文件偏移量
    iunlock(f->ip);
  } else {
    panic("fileread");
  }

  return r;
}

3.5 filewrite — 统一写接口（含分批写）

int
filewrite(struct file *f, uint64 addr, int n)
{
  int r, ret = 0;

  if(f->writable == 0)
    return -1;

  if(f->type == FD_PIPE){
    ret = pipewrite(f->pipe, addr, n);
  } else if(f->type == FD_DEVICE){
    if(f->major < 0 || f->major >= NDEV || !devsw[f->major].write)
      return -1;
    ret = devsw[f->major].write(1, addr, n);
  } else if(f->type == FD_INODE){
    // 分批写，防止单次 log 事务超出 MAXOPBLOCKS
    // max = 每批最大字节数:
    //   MAXOPBLOCKS=10, 减去 inode块/间接块/分配块/2块余量 = (10-1-1-2)/2 * 1024 = 3072 字节
    int max = ((MAXOPBLOCKS-1-1-2) / 2) * BSIZE;
    int i = 0;
    while(i < n){
      int n1 = n - i;
      if(n1 > max)
        n1 = max;

      begin_op();               // 开启 log 事务
      ilock(f->ip);
      if((r = writei(f->ip, 1, addr + i, f->off, n1)) > 0)
        f->off += r;
      iunlock(f->ip);
      end_op();                 // 提交 log 事务

      if(r < 0)
        break;
      if(r != n1)
        panic("short filewrite");
      i += r;
    }
    ret = (i == n ? n : -1);
  } else {
    panic("filewrite");
  }

  return ret;
}

分批写的必要性：写磁盘文件时，每次 writei 可能触发块分配、更新 inode、更新间接块等多个日志操作。若一次写入太大（超出 MAXOPBLOCKS 限制），会导致 log_write panic。所以每批最多写 3072 字节（3块），确保在日志容量内完成。

四、管道（pipe）深度解析

4.1 数据结构

// kernel/pipe.c
#define PIPESIZE 512

struct pipe {
  struct spinlock lock;
  char data[PIPESIZE];   // 512 字节环形缓冲区
  uint nread;            // 已读总字节数（单调递增）
  uint nwrite;           // 已写总字节数（单调递增）
  int readopen;          // 读端是否仍打开
  int writeopen;         // 写端是否仍打开
};

环形缓冲区设计：

data[PIPESIZE] 数组：

索引:   0   1   2   ...  511
        ↑               ↑
   nread%512         nwrite%512

缓冲区满：nwrite == nread + PIPESIZE  (差值 = 512)
缓冲区空：nwrite == nread             (差值 = 0)

当前数据量：nwrite - nread
空闲空间：  PIPESIZE - (nwrite - nread)

无需清零字节，nread 和 nwrite 单调递增，取模 PIPESIZE 做下标，天然处理绕回。

4.2 pipealloc — 创建管道

int
pipealloc(struct file **f0, struct file **f1)
{
  struct pipe *pi;

  pi = 0;
  *f0 = *f1 = 0;
  if((*f0 = filealloc()) == 0 || (*f1 = filealloc()) == 0)
    goto bad;
  if((pi = (struct pipe*)kalloc()) == 0)   // 分配一页内存存放 pipe 结构
    goto bad;

  pi->readopen = 1;
  pi->writeopen = 1;
  pi->nwrite = 0;
  pi->nread = 0;
  initlock(&pi->lock, "pipe");

  (*f0)->type = FD_PIPE;
  (*f0)->readable = 1;   // f0 是读端
  (*f0)->writable = 0;
  (*f0)->pipe = pi;

  (*f1)->type = FD_PIPE;
  (*f1)->readable = 0;
  (*f1)->writable = 1;   // f1 是写端
  (*f1)->pipe = pi;
  return 0;

 bad:
  if(pi)  kfree((char*)pi);
  if(*f0) fileclose(*f0);
  if(*f1) fileclose(*f1);
  return -1;
}

一个管道对应两个 struct file：读端 f0 和写端 f1，共享同一个 struct pipe。

4.3 pipewrite — 写管道

int
pipewrite(struct pipe *pi, uint64 addr, int n)
{
  int i;
  char ch;
  struct proc *pr = myproc();

  acquire(&pi->lock);
  for(i = 0; i < n; i++){
    // 缓冲区满 → sleep 等待读者消费
    while(pi->nwrite == pi->nread + PIPESIZE){
      if(pi->readopen == 0 || pr->killed){
        release(&pi->lock);
        return -1;           // 读端已关闭或进程被杀死
      }
      wakeup(&pi->nread);    // 唤醒等待读的进程
      sleep(&pi->nwrite, &pi->lock);   // 在 nwrite 上睡眠
    }
    // 从用户空间复制1字节到缓冲区
    if(copyin(pr->pagetable, &ch, addr + i, 1) == -1)
      break;
    pi->data[pi->nwrite++ % PIPESIZE] = ch;
  }
  wakeup(&pi->nread);        // 通知等待读者
  release(&pi->lock);
  return i;
}

4.4 piperead — 读管道

int
piperead(struct pipe *pi, uint64 addr, int n)
{
  int i;
  struct proc *pr = myproc();
  char ch;

  acquire(&pi->lock);
  // 缓冲区空且写端仍打开 → sleep 等待
  while(pi->nread == pi->nwrite && pi->writeopen){
    if(pr->killed){
      release(&pi->lock);
      return -1;
    }
    sleep(&pi->nread, &pi->lock);   // 在 nread 上睡眠
  }
  // 读尽可能多的字节（最多 n 字节）
  for(i = 0; i < n; i++){
    if(pi->nread == pi->nwrite)
      break;
    ch = pi->data[pi->nread++ % PIPESIZE];
    if(copyout(pr->pagetable, addr + i, &ch, 1) == -1)
      break;
  }
  wakeup(&pi->nwrite);     // 通知等待的写者
  release(&pi->lock);
  return i;
}

sleep/wakeup 配对关系：

写者 sleep 在 &pi->nwrite   ←→  读者 wakeup(&pi->nwrite)
读者 sleep 在 &pi->nread    ←→  写者 wakeup(&pi->nread)

4.5 pipeclose — 关闭管道端

void
pipeclose(struct pipe *pi, int writable)
{
  acquire(&pi->lock);
  if(writable){
    pi->writeopen = 0;
    wakeup(&pi->nread);    // 唤醒读者，使其发现写端已关
  } else {
    pi->readopen = 0;
    wakeup(&pi->nwrite);   // 唤醒写者，使其返回 -1
  }
  if(pi->readopen == 0 && pi->writeopen == 0){
    release(&pi->lock);
    kfree((char*)pi);      // 两端都关闭，释放内存
  } else
    release(&pi->lock);
}

关闭语义：当写端关闭后，读者从阻塞中醒来，发现 pi->writeopen == 0 且缓冲区中仍有数据，则继续读完剩余数据；数据读完后，nread == nwrite && !writeopen，退出循环返回 0（EOF）。

五、sysfile.c 系统调用层精读

系统调用层承担两个职责：（1）安全地从用户空间提取参数；（2）调用内核内部函数完成操作。

5.1 辅助函数 argfd / fdalloc

// 从系统调用第 n 个参数提取文件描述符及对应 file 指针
static int
argfd(int n, int *pfd, struct file **pf)
{
  int fd;
  struct file *f;

  if(argint(n, &fd) < 0)              // 读取第 n 个整型参数
    return -1;
  // 边界检查：fd 范围有效, 且进程的 ofile[fd] 不为空
  if(fd < 0 || fd >= NOFILE || (f=myproc()->ofile[fd]) == 0)
    return -1;
  if(pfd) *pfd = fd;
  if(pf)  *pf  = f;
  return 0;
}

// 在进程 ofile[] 中找第一个空闲槽位，注册 file 指针
static int
fdalloc(struct file *f)
{
  int fd;
  struct proc *p = myproc();

  for(fd = 0; fd < NOFILE; fd++){
    if(p->ofile[fd] == 0){
      p->ofile[fd] = f;
      return fd;
    }
  }
  return -1;    // 进程打开文件数已达上限 NOFILE=16
}

5.2 sys_open — 打开/创建文件

sys_open 是最复杂的系统调用，涵盖参数解析 → 路径查找/创建 → 分配内核对象 → 返回 fd 的完整流程：

uint64
sys_open(void)
{
  char path[MAXPATH];
  int fd, omode;
  struct file *f;
  struct inode *ip;
  int n;

  // 1. 参数提取：路径字符串 + 打开模式
  if((n = argstr(0, path, MAXPATH)) < 0 || argint(1, &omode) < 0)
    return -1;

  begin_op();   // 开启日志事务

  // 2. 路径解析: 创建 or 查找
  if(omode & O_CREATE){
    ip = create(path, T_FILE, 0, 0);   // 内部调用 nameiparent + ialloc + dirlink
    if(ip == 0){ end_op(); return -1; }
  } else {
    if((ip = namei(path)) == 0){ end_op(); return -1; }
    ilock(ip);
    // 目录只允许 O_RDONLY 打开
    if(ip->type == T_DIR && omode != O_RDONLY){
      iunlockput(ip); end_op(); return -1;
    }
  }

  // 3. 设备号合法性检查
  if(ip->type == T_DEVICE && (ip->major < 0 || ip->major >= NDEV)){
    iunlockput(ip); end_op(); return -1;
  }

  // 4. 分配 struct file + fd
  if((f = filealloc()) == 0 || (fd = fdalloc(f)) < 0){
    if(f) fileclose(f);
    iunlockput(ip); end_op(); return -1;
  }

  // 5. 初始化 struct file 字段
  if(ip->type == T_DEVICE){
    f->type = FD_DEVICE;
    f->major = ip->major;   // 记录主设备号用于 devsw 分发
  } else {
    f->type = FD_INODE;
    f->off = 0;             // 文件偏移从 0 开始
  }
  f->ip = ip;
  f->readable = !(omode & O_WRONLY);
  f->writable = (omode & O_WRONLY) || (omode & O_RDWR);

  // 6. O_TRUNC：截断文件内容
  if((omode & O_TRUNC) && ip->type == T_FILE){
    itrunc(ip);
  }

  iunlock(ip);
  end_op();   // 提交日志事务

  return fd;
}

sys_open 调用链图：

sys_open
  ├── argstr / argint           提取参数
  ├── begin_op                  开启日志事务
  ├── create(path, T_FILE)      (O_CREATE 情况)
  │   ├── nameiparent           找父目录 inode
  │   ├── dirlookup             检查是否已存在
  │   ├── ialloc                分配磁盘 inode
  │   ├── ilock / iupdate       初始化 inode
  │   └── dirlink               在父目录添加目录项
  ├── namei(path)               (普通 open 情况)
  ├── filealloc                 从 ftable 分配 struct file
  ├── fdalloc                   在 proc->ofile[] 分配 fd
  ├── itrunc                    (O_TRUNC 情况)
  ├── iunlock / end_op          提交日志事务
  └── return fd

5.3 sys_read / sys_write / sys_close

这三个调用相对简单，是对 file.c 函数的薄包装：

uint64
sys_read(void)
{
  struct file *f;
  int n;
  uint64 p;

  // 参数：fd(0), buf_addr(1), n_bytes(2)
  if(argfd(0, 0, &f) < 0 || argint(2, &n) < 0 || argaddr(1, &p) < 0)
    return -1;
  return fileread(f, p, n);    // 直接委托给 fileread
}

uint64
sys_write(void)
{
  struct file *f;
  int n;
  uint64 p;

  if(argfd(0, 0, &f) < 0 || argint(2, &n) < 0 || argaddr(1, &p) < 0)
    return -1;
  return filewrite(f, p, n);   // 直接委托给 filewrite
}

uint64
sys_close(void)
{
  int fd;
  struct file *f;

  if(argfd(0, &fd, &f) < 0)
    return -1;
  myproc()->ofile[fd] = 0;     // 从进程 fd 表中移除
  fileclose(f);                // 减引用计数（可能真正关闭）
  return 0;
}

5.4 sys_dup

uint64
sys_dup(void)
{
  struct file *f;
  int fd;

  if(argfd(0, 0, &f) < 0)
    return -1;
  if((fd=fdalloc(f)) < 0)    // 新 fd 指向同一 struct file
    return -1;
  filedup(f);                // 增加 ref 计数
  return fd;
}

dup(oldfd) 后两个 fd 共享 struct file（包括共享 off 偏移量），这是 shell 中 2>&1 重定向的实现基础。

5.5 sys_link / sys_unlink — 硬链接

sys_link：让 new 路径成为 old 文件的另一个硬链接。

uint64
sys_link(void)
{
  char name[DIRSIZ], new[MAXPATH], old[MAXPATH];
  struct inode *dp, *ip;

  if(argstr(0, old, MAXPATH) < 0 || argstr(1, new, MAXPATH) < 0)
    return -1;

  begin_op();
  if((ip = namei(old)) == 0){ end_op(); return -1; }

  ilock(ip);
  if(ip->type == T_DIR){           // 不允许对目录建硬链接
    iunlockput(ip); end_op(); return -1;
  }

  ip->nlink++;                     // 磁盘 inode 的 nlink 加 1
  iupdate(ip);                     // 写回磁盘
  iunlock(ip);

  if((dp = nameiparent(new, name)) == 0) goto bad;
  ilock(dp);
  if(dp->dev != ip->dev || dirlink(dp, name, ip->inum) < 0){
    iunlockput(dp); goto bad;
  }
  iunlockput(dp);
  iput(ip);
  end_op();
  return 0;
  ...
}

sys_unlink：删除目录项，若 nlink 降为 0 且无进程打开该文件，则释放 inode 和数据块。

// 核心逻辑（省略参数解析）
ip->nlink--;
iupdate(ip);          // 写回磁盘的 nlink 变化
// 目录项置零（inum=0）即删除
memset(&de, 0, sizeof(de));
writei(dp, 0, (uint64)&de, off, sizeof(de));

5.6 create — 内部文件/目录创建

create 是 sys_open(O_CREATE), sys_mkdir, sys_mknod 共享的内部辅助函数：

static struct inode*
create(char *path, short type, short major, short minor)
{
  struct inode *ip, *dp;
  char name[DIRSIZ];

  if((dp = nameiparent(path, name)) == 0)
    return 0;

  ilock(dp);

  // 幂等性：若已存在且是 T_FILE，直接返回（open O_CREATE 的语义）
  if((ip = dirlookup(dp, name, 0)) != 0){
    iunlockput(dp);
    ilock(ip);
    if(type == T_FILE && (ip->type == T_FILE || ip->type == T_DEVICE))
      return ip;       // 文件已存在，视为成功（不截断，O_TRUNC 之后处理）
    iunlockput(ip);
    return 0;
  }

  // 分配新 inode
  if((ip = ialloc(dp->dev, type)) == 0)
    panic("create: ialloc");

  ilock(ip);
  ip->major = major;
  ip->minor = minor;
  ip->nlink = 1;
  iupdate(ip);

  if(type == T_DIR){
    dp->nlink++;       // 子目录 ".." 指向父目录，父目录 nlink 增加
    iupdate(dp);
    if(dirlink(ip, ".", ip->inum) < 0 || dirlink(ip, "..", dp->inum) < 0)
      panic("create dots");
  }

  if(dirlink(dp, name, ip->inum) < 0)
    panic("create: dirlink");

  iunlockput(dp);
  return ip;           // 返回时 ip 仍持有锁，由调用者解锁
}

5.7 sys_pipe

uint64
sys_pipe(void)
{
  uint64 fdarray;     // 用户空间存放 [fd0, fd1] 的数组地址
  struct file *rf, *wf;
  int fd0, fd1;
  struct proc *p = myproc();

  if(argaddr(0, &fdarray) < 0)
    return -1;
  if(pipealloc(&rf, &wf) < 0)     // 创建管道，分配两个 struct file
    return -1;

  fd0 = -1;
  if((fd0 = fdalloc(rf)) < 0 || (fd1 = fdalloc(wf)) < 0){
    // 回滚：释放已分配的 fd 和 struct file
    if(fd0 >= 0) p->ofile[fd0] = 0;
    fileclose(rf);
    fileclose(wf);
    return -1;
  }
  // 将 [fd0, fd1] 写回用户空间数组
  if(copyout(p->pagetable, fdarray,               (char*)&fd0, sizeof(fd0)) < 0 ||
     copyout(p->pagetable, fdarray+sizeof(fd0),   (char*)&fd1, sizeof(fd1)) < 0){
    p->ofile[fd0] = 0;
    p->ofile[fd1] = 0;
    fileclose(rf);
    fileclose(wf);
    return -1;
  }
  return 0;
}

5.8 sys_chdir

uint64
sys_chdir(void)
{
  char path[MAXPATH];
  struct inode *ip;
  struct proc *p = myproc();
  
  begin_op();
  if(argstr(0, path, MAXPATH) < 0 || (ip = namei(path)) == 0){
    end_op(); return -1;
  }
  ilock(ip);
  if(ip->type != T_DIR){           // 必须是目录
    iunlockput(ip); end_op(); return -1;
  }
  iunlock(ip);
  iput(p->cwd);    // 释放旧的工作目录
  end_op();
  p->cwd = ip;     // 更新进程的当前工作目录
  return 0;
}

六、完整数据流：write(fd, buf, n) 系统调用路径

以向普通文件写入数据为例，展示从用户空间到磁盘的完整路径：

用户程序
  write(fd, buf, n)
        │ ecall
        ↓
  usertrap() → syscall() → sys_write()
        │ argfd(0,0,&f)  argaddr(1,&p)  argint(2,&n)
        ↓
  filewrite(f, addr, n)
        │ f->type == FD_INODE
        │ 分批（每批 ≤ 3072 字节）
        ↓
  begin_op()                  ← 开启日志事务
  ilock(f->ip)
  writei(f->ip, 1, addr+i, f->off, n1)
        │ bmap(ip, blockno)   ← 找到/分配数据块号
        │ bread(dev, blockno) ← 读入 buf cache
        │ copyin(pagetable, buf, addr, n1) ← 用户→内核复制
        │ log_write(bp)       ← 标记该块为"脏"，加入 log
        │ brelse(bp)
        │ f->off += n1
  iunlock(f->ip)
  end_op()                    ← 提交日志（写 log header → 写 log 数据 → 写 checksum）
        │
        ↓
  磁盘（VirtIO DMA 写入）

参与层次：

系统调用层  sysfile.c    sys_write  — 参数校验
文件对象层  file.c       filewrite  — 类型分发 + 分批逻辑
日志层      log.c        begin_op/end_op/log_write  — 崩溃一致性
Inode层     fs.c         writei/bmap  — 块地址映射
块缓存层    bio.c        bread/bwrite/brelse  — LRU 缓存
磁盘驱动    virtio_disk.c  — DMA 传输

七、fork 时的文件描述符继承

fork() 在 kernel/proc.c 中处理：

// 复制父进程的文件描述符表
for(i = 0; i < NOFILE; i++)
  if(p->ofile[i])
    np->ofile[i] = filedup(p->ofile[i]);   // 共享 struct file，ref 加 1
np->cwd = idup(p->cwd);                    // 共享工作目录 inode

fork 后的引用结构：

父进程 ofile[1] ──┐
                  ├──→ struct file { ref=2, FD_INODE, off=100 }
子进程 ofile[1] ──┘         │
                          struct inode { ref=1, ... }

父子进程共享同一个 struct file 对象（包括 off），这确保了在 shell 管道中，父子进程对同一文件的读写位置是同步的。

实验一：观察文件描述符分配

#include "kernel/types.h"

    close(fd1);
    int fd3 = open("/README", O_RDONLY);

    close(fd3);

}

实验二：管道读写验证

// user/pipetest.c (自行创建)
#include "kernel/types.h"
#include "user/user.h"

int main() {
    int p[2];
    pipe(p);  // p[0]=读端, p[1]=写端

    if(fork() == 0) {
        // 子进程：写管道
        close(p[0]);
        char *msg = "hello from child\n";
        write(p[1], msg, strlen(msg));
        close(p[1]);
        exit(0);
    } else {
        // 父进程：读管道
        close(p[1]);
        char buf[64];
        int n = read(p[0], buf, sizeof(buf));
        buf[n] = 0;
        printf("parent got: %s", buf);
        close(p[0]);
        wait(0);
    }
    exit(0);
}

实验三：硬链接验证

# 在 xv6 shell 中
$ echo hello > /tmp/a
$ ln /tmp/a /tmp/b         # 创建硬链接
$ cat /tmp/b               # 输出: hello
$ rm /tmp/a                # 删除 a (nlink: 2→1)
$ cat /tmp/b               # 数据仍可访问
$ rm /tmp/b                # 删除 b (nlink: 1→0, inode 释放)

验证要点：nlink 降为 0 且无进程打开时，iput 才调用 itrunc 释放数据块。

实验四：追踪 sys_open 的 log 事务

在 kernel/log.c 的 commit() 函数中添加打印：

static void
commit()
{
  if (log.lh.n > 0) {
    printf("commit: writing %d blocks\n", log.lh.n);  // 添加这行
    write_log();
    write_head();
    install_trans(0);
    log.lh.n = 0;
    write_head();
  }
}

重新编译后，执行 echo hello > /test 会看到类似输出：

commit: writing 3 blocks

说明一次文件创建+写入触发了 3 个块的日志提交（inode 块、目录块、数据块）。

实验五：观察管道在缓冲区满时的阻塞

分析 pipewrite 中的 while 循环：当 nwrite - nread == 512 时，写者 sleep，直到读者消费数据。用 ps 或调试输出可以观察到进程状态变化（运行→睡眠→运行）。

九、本节小结

组件 位置 核心作用
struct file kernel/file.h 内核文件对象，持有 type/ref/off/ip
ftable kernel/file.c 全局文件表（最多 100 个）
proc->ofile[] kernel/proc.h 进程 fd 表（最多 16 个）
filealloc/filedup/fileclose kernel/file.c 引用计数管理
fileread/filewrite kernel/file.c 统一读写，按 type 分发
devsw[] kernel/file.c 设备函数指针表（多态）
struct pipe kernel/pipe.c 512 字节环形缓冲区 + sleep/wakeup
pipealloc/read/write/close kernel/pipe.c 管道生命周期
argfd / fdalloc kernel/sysfile.c 参数提取辅助
sys_open kernel/sysfile.c 最复杂的文件系统调用
create() kernel/sysfile.c 内部辅助：创建文件/目录/设备
sys_link/unlink kernel/sysfile.c 硬链接管理

核心设计原则：

  1. 两层抽象：进程 fd 表（轻量整数）+ 全局文件表（含状态）
  2. 多态分发：fileread/filewrite 屏蔽底层差异，通过 type 或 devsw 分发
  3. 引用计数：struct file 的 ref 计数确保 fork/dup 后安全共享
  4. 日志保护：文件写入封装在 begin_op/end_op 中，保证崩溃一致性
  5. 睡眠等待：管道用 sleep/wakeup 实现背压（backpressure），避免忙等
