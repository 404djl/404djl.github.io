第 15 节：Inode 层与目录

学习目标：理解 xv6 文件系统中 inode 的磁盘结构与内存表示；掌握块分配（balloc/bfree）、inode 缓存操作（iget/ilock/iput）、数据读写（bmap/readi/writei）的完整实现；深入分析目录操作（dirlookup/dirlink）和路径解析（namex）的递归逻辑。

预计用时：4-5 小时（fs.c 是本课程最大的单文件，671 行）

前置要求：完成第 13 节（块缓存：bread/bwrite/brelse）和第 14 节（日志层：log_write/begin_op/end_op）；理解 sleeplock 的使用方式（第 12 节）

一、理论：文件系统分层与 inode 的核心概念

1.1 本节在文件系统中的位置

第 5 层  路径名（pathname）     ← namei(), nameiparent()     本节
第 4 层  目录（directory）      ← dirlookup(), dirlink()     本节
第 3 层  Inode（inode）         ← ialloc(), readi(), writei()本节
第 2 层  块分配（block alloc）   ← balloc(), bfree()          本节
─────────────── 以上均在 kernel/fs.c ─────────────────────────
第 1 层  日志（logging）         ← begin_op(), log_write()    第 14 节
第 0 层  块缓存（buffer cache）  ← bread(), bwrite()          第 13 节

1.2 Inode 是什么？

Inode（Index Node，索引节点）是文件的元数据容器，它描述一个文件的”属性”，并通过块地址数组指向文件的实际数据。

一个 inode 不包含文件名——文件名存储在目录中。inode 只关心”文件有什么内容”，目录负责回答”这个名字对应哪个 inode”。

文件名（在目录中）      目录条目             inode                    数据块
"README"          →   {inum=5, name="README"} → inode #5 → [块45][块46][块47]

1.3 磁盘上的 inode：struct dinode

// kernel/fs.h

struct dinode {
  short type;            // 文件类型：0=未分配, T_FILE=2, T_DIR=1, T_DEVICE=3
  short major;           // 主设备号（仅 T_DEVICE）
  short minor;           // 次设备号（仅 T_DEVICE）
  short nlink;           // 硬链接数（多少个目录条目指向本 inode）
  uint  size;            // 文件大小（字节数）
  uint  addrs[NDIRECT+1]; // 数据块地址数组（12 个直接 + 1 个间接）
};

#define NDIRECT   12
#define NINDIRECT (BSIZE / sizeof(uint))   // = 1024/4 = 256
#define MAXFILE   (NDIRECT + NINDIRECT)    // = 12 + 256 = 268 块

块地址数组的混合索引结构：

ip->addrs[0]  ~ ip->addrs[11]   直接块号（12个）
ip->addrs[12]                   一级间接块号
                                  └─ 间接块内存放 256 个数据块号

最大文件大小 = (12 + 256) × 1024 字节 = 268 KB

磁盘 inode 布局：

// 每个块能存储的 inode 数
#define IPB  (BSIZE / sizeof(struct dinode))   // = 1024 / 64 = 16

// inode i 所在的磁盘块号
#define IBLOCK(i, sb)  ((i) / IPB + sb.inodestart)

例如 inode #5：所在块 = 5/16 + inodestart = 0 + 32（inodestart=32）= 块 32；块内偏移 = 5%16 = 第 5 个 dinode。

1.4 内存中的 inode：struct inode

// kernel/file.h

struct inode {
  uint dev;            // 设备号（与对应磁盘块上的设备一致）
  uint inum;           // inode 编号（1-based，对应磁盘上的位置）
  int  ref;            // 内存引用计数（有多少指针指向此 inode）
  struct sleeplock lock; // 保护以下所有字段（valid 及之后）

  int   valid;         // 是否已从磁盘读入（0 = 未读，1 = 已读）
  // 以下字段是磁盘 dinode 的内存副本：
  short type;
  short major;
  short minor;
  short nlink;
  uint  size;
  uint  addrs[NDIRECT+1];
};

两层计数的区别：

字段 含义 由谁保护
ip->ref 内存引用计数：有多少 struct inode * 指针在使用 icache.lock（自旋锁）
ip->nlink 磁盘链接数：有多少目录条目硬链接到此 inode ip->lock（睡眠锁）

ref == 0：inode 缓存条目空闲，可以被复用
nlink == 0 且 ref == 0：文件彻底删除，iput() 会截断并释放磁盘 inode

1.5 inode 缓存（icache）

struct {
  struct spinlock lock;         // 保护 icache 条目分配
  struct inode inode[NINODE];   // NINODE = 50 个缓存槽
} icache;

inode 缓存是一个固定大小的数组（不是像 bcache 那样的 LRU 链表）。通过 ref 引用计数管理生命周期。

二、源码精读：块分配层

2.1 balloc()——分配一个空闲数据块

static uint balloc(uint dev)
{
  int b, bi, m;
  struct buf *bp;

  bp = 0;
  for(b = 0; b < sb.size; b += BPB) {
    // BPB = 位图每块能表示的块数 = BSIZE*8 = 8192
    bp = bread(dev, BBLOCK(b, sb));   // 读位图块（BBLOCK 计算位图块号）
    for(bi = 0; bi < BPB && b + bi < sb.size; bi++) {
      m = 1 << (bi % 8);              // 当前位的掩码
      if((bp->data[bi/8] & m) == 0) { // 该位为 0 = 空闲
        bp->data[bi/8] |= m;          // 置 1 = 标记已用
        log_write(bp);                 // 写入日志（事务保护）
        brelse(bp);
        bzero(dev, b + bi);            // 将新分配的块清零（安全）
        return b + bi;                 // 返回块号
      }
    }
    brelse(bp);
  }
  panic("balloc: out of blocks");
}

位图设计：每个数据块对应位图中的一个 bit（0=空闲，1=已用）。BBLOCK(b, sb) 计算块 b 对应的位图块号：

#define BPB      (BSIZE * 8)                        // 每个位图块表示 8192 个块
#define BBLOCK(b, sb)  ((b)/BPB + sb.bmapstart)     // 块 b 在哪个位图块

bzero() 在分配后将新块清零：防止旧数据泄露给新文件（信息安全考虑）。

2.2 bfree()——释放一个数据块

static void bfree(int dev, uint b)
{
  struct buf *bp;
  int bi, m;

  bp = bread(dev, BBLOCK(b, sb));  // 读位图块
  bi = b % BPB;                    // 块 b 在位图块内的 bit 偏移
  m  = 1 << (bi % 8);             // 掩码
  if((bp->data[bi/8] & m) == 0)
    panic("freeing free block");   // 重复释放检测
  bp->data[bi/8] &= ~m;           // 清 0 = 标记为空闲
  log_write(bp);
  brelse(bp);
}

注意：bfree() 只清除位图，不清零块内容（安全建议：应先清零，但 xv6 为简洁省略）。

三、源码精读：inode 操作

3.1 iinit()——初始化 inode 缓存

void iinit()
{
  initlock(&icache.lock, "icache");
  for(int i = 0; i < NINODE; i++)
    initsleeplock(&icache.inode[i].lock, "inode");
}

3.2 ialloc()——在磁盘上分配新 inode

struct inode* ialloc(uint dev, short type)
{
  int inum;
  struct buf *bp;
  struct dinode *dip;

  for(inum = 1; inum < sb.ninodes; inum++) {
    // 遍历所有 inode（inode 编号从 1 开始，0 是无效值）
    bp  = bread(dev, IBLOCK(inum, sb));       // 读包含 inum 的 inode 块
    dip = (struct dinode*)bp->data + inum%IPB; // 块内偏移

    if(dip->type == 0) {   // type == 0 表示未分配
      memset(dip, 0, sizeof(*dip));  // 清零整个 dinode
      dip->type = type;              // 设置文件类型（占用该 inode）
      log_write(bp);                 // 日志记录分配操作
      brelse(bp);
      return iget(dev, inum);        // 将 inode 加入内存缓存并返回
    }
    brelse(bp);
  }
  panic("ialloc: no inodes");  // 磁盘 inode 区耗尽
}

ialloc() 返回未锁定但已引用的 inode。注释里说 “Returns an unlocked but allocated and referenced inode”——这很重要：调用者得到一个 ref > 0 的 inode，但需要自己调用 ilock() 才能读写它。

3.3 iget()——查找或创建内存 inode（不加锁）

static struct inode* iget(uint dev, uint inum)
{
  struct inode *ip, *empty;

  acquire(&icache.lock);

  // ① 检查是否已缓存（有 ref > 0 的同 dev/inum 条目）
  empty = 0;
  for(ip = &icache.inode[0]; ip < &icache.inode[NINODE]; ip++) {
    if(ip->ref > 0 && ip->dev == dev && ip->inum == inum) {
      ip->ref++;                  // 已缓存，增加引用计数
      release(&icache.lock);
      return ip;                  // 直接返回（不从磁盘读，valid 可能已是 1）
    }
    if(empty == 0 && ip->ref == 0)
      empty = ip;                 // 记录空闲槽位（备用）
  }

  // ② 未缓存，使用空闲槽位
  if(empty == 0)
    panic("iget: no inodes");

  ip = empty;
  ip->dev   = dev;
  ip->inum  = inum;
  ip->ref   = 1;
  ip->valid = 0;    // 数据尚未从磁盘读入
  release(&icache.lock);

  return ip;  // 返回已引用但未锁定、未读磁盘的 inode
}

iget() 只处理内存引用，不做磁盘 I/O。这是一个重要的设计选择：在路径遍历时，我们可能会查找很多 inode，但不一定需要读取所有 inode 的内容——只需要记住”存在”并增加引用计数，真正需要内容时才 ilock()。

**iget() vs idup()**：

struct inode* idup(struct inode *ip)
{
  acquire(&icache.lock);
  ip->ref++;    // 已有指针，只增加引用计数
  release(&icache.lock);
  return ip;
}
  • iget(dev, inum)：通过设备号和 inode 编号查找（或新建）缓存条目
  • idup(ip)：已有 inode 指针，仅增加引用计数（如 fork 复制 cwd）

3.4 ilock()——锁定 inode 并从磁盘读取

void ilock(struct inode *ip)
{
  struct buf *bp;
  struct dinode *dip;

  if(ip == 0 || ip->ref < 1)
    panic("ilock");

  acquiresleep(&ip->lock);   // 获取 sleeplock（可能阻塞）

  if(ip->valid == 0) {
    // 尚未从磁盘读入，现在读
    bp  = bread(ip->dev, IBLOCK(ip->inum, sb));
    dip = (struct dinode*)bp->data + ip->inum % IPB;
    ip->type  = dip->type;
    ip->major = dip->major;
    ip->minor = dip->minor;
    ip->nlink = dip->nlink;
    ip->size  = dip->size;
    memmove(ip->addrs, dip->addrs, sizeof(ip->addrs));
    brelse(bp);
    ip->valid = 1;
    if(ip->type == 0)
      panic("ilock: no type");  // type == 0 表示 inode 未分配（错误）
  }
}

ilock() 与 iget() 分离的意义：

许多操作需要持有 inode 引用（防止被释放）但不需要锁定内容，例如：

  • open() 打开文件后，文件描述符持有 ip->ref++，但只有 read/write 时才短暂 ilock()
  • 路径解析遍历时，持有父目录 inode 的引用，但可以与子 inode 分别锁

如果 iget() 本身就加锁，路径遍历时就会同时持有多把 sleeplock，死锁风险大增。

3.5 iunlock() 和 iput()——解锁与释放引用

void iunlock(struct inode *ip)
{
  if(ip == 0 || !holdingsleep(&ip->lock) || ip->ref < 1)
    panic("iunlock");
  releasesleep(&ip->lock);
}

void iput(struct inode *ip)
{
  acquire(&icache.lock);

  if(ip->ref == 1 && ip->valid && ip->nlink == 0) {
    // ref 将变为 0，且没有目录链接 → 文件被删除，释放磁盘资源
    // ref == 1 意味着没有其他进程持有此 inode，acquiresleep 不会阻塞
    acquiresleep(&ip->lock);
    release(&icache.lock);

    itrunc(ip);     // 释放所有数据块
    ip->type = 0;   // type = 0 表示 inode 空闲
    iupdate(ip);    // 将 type=0 写入磁盘（即释放磁盘 inode）
    ip->valid = 0;

    releasesleep(&ip->lock);
    acquire(&icache.lock);
  }

  ip->ref--;
  release(&icache.lock);
}

**iput() 中的”延迟删除”**：

当文件被 unlink() 但还有进程打开时（如 rm 一个正在被 cat 读的文件）：

  • nlink 减到 0（目录中无链接）
  • ref 仍 > 0（进程持有文件描述符）
  • iput() 此时不删除，因为 ref > 1

直到最后一个 close() 使 ref 减到 1，调用 iput() 时才真正释放磁盘资源。这是 Unix “打开的文件不会消失” 语义的实现基础。

**常用组合：iunlockput(ip)**：

void iunlockput(struct inode *ip)
{
  iunlock(ip);
  iput(ip);
}

iunlock + iput 频繁组合使用，xv6 提供这个便利函数。

3.6 iupdate()——将内存 inode 同步回磁盘

void iupdate(struct inode *ip)
{
  struct buf *bp;
  struct dinode *dip;

  bp  = bread(ip->dev, IBLOCK(ip->inum, sb));
  dip = (struct dinode*)bp->data + ip->inum % IPB;
  dip->type  = ip->type;
  dip->major = ip->major;
  dip->minor = ip->minor;
  dip->nlink = ip->nlink;
  dip->size  = ip->size;
  memmove(dip->addrs, ip->addrs, sizeof(ip->addrs));
  log_write(bp);  // 通过日志写回（事务保护）
  brelse(bp);
}

iupdate() 是写透（write-through）：每次在内存中修改 inode 字段（type、size、addrs[] 等），都立即调用 iupdate() 将变更写入日志。这保证了崩溃时不会丢失 inode 元数据的修改。

四、源码精读：文件数据读写

4.1 bmap()——逻辑块号 → 物理块号（必要时分配）

static uint bmap(struct inode *ip, uint bn)
{
  uint addr, *a;
  struct buf *bp;

  // ① 直接块（0 ~ NDIRECT-1）
  if(bn < NDIRECT) {
    if((addr = ip->addrs[bn]) == 0)
      ip->addrs[bn] = addr = balloc(ip->dev);  // 按需分配
    return addr;
  }
  bn -= NDIRECT;

  // ② 一级间接块（NDIRECT ~ NDIRECT+NINDIRECT-1）
  if(bn < NINDIRECT) {
    // 加载（或分配）间接块本身
    if((addr = ip->addrs[NDIRECT]) == 0)
      ip->addrs[NDIRECT] = addr = balloc(ip->dev);

    bp = bread(ip->dev, addr);       // 读间接块
    a  = (uint*)bp->data;            // 间接块内容是 256 个块号

    if((addr = a[bn]) == 0) {
      a[bn] = addr = balloc(ip->dev);  // 按需分配数据块
      log_write(bp);                   // 间接块被修改，写日志
    }
    brelse(bp);
    return addr;
  }

  panic("bmap: out of range");
}

bmap() 的关键特性——按需分配（demand allocation）：

bmap() 在块不存在时自动分配，这使得 readi/writei 可以统一调用 bmap() 处理所有情况，不需要区分”块已存在”和”块需要创建”。

间接块的两级磁盘访问：对于大文件的末尾数据（块号 ≥ 12），需要：

  1. 读间接块（1 次磁盘读）
  2. 读间接块内记录的数据块号，再读数据块（1 次磁盘读）

总共比直接块多 1 次磁盘 I/O。

4.2 readi()——从 inode 读数据

int readi(struct inode *ip, int user_dst, uint64 dst, uint off, uint n)
{
  uint tot, m;
  struct buf *bp;

  if(off > ip->size || off + n < off)
    return 0;
  if(off + n > ip->size)
    n = ip->size - off;   // 截断到文件末尾

  for(tot=0; tot<n; tot+=m, off+=m, dst+=m) {
    bp = bread(ip->dev, bmap(ip, off/BSIZE));  // 读包含 off 的块
    m  = min(n - tot, BSIZE - off%BSIZE);      // 本次从此块读取的字节数
    if(either_copyout(user_dst, dst, bp->data + (off % BSIZE), m) == -1) {
      brelse(bp);
      break;
    }
    brelse(bp);
  }
  return tot;
}

循环逻辑解析：

读取 [off, off+n) 字节：

第一次迭代：
  块号 = off / BSIZE（off 所在的块）
  m    = min(n, BSIZE - off%BSIZE)（到块边界的字节数）
  从 bp->data[off%BSIZE] 位置开始读 m 字节

中间迭代：
  块号 = (aligned) off / BSIZE（完整块）
  m    = min(remaining, BSIZE) = BSIZE（通常读满整块）

最后一次迭代：
  块号 = (off+n-1) / BSIZE
  m    = 剩余字节（可能不满整块）

**either_copyout()**：根据 user_dst 标志，将数据拷贝到用户空间（copyout）或内核空间（memmove）。这使 readi() 能同时服务用户程序（sys_read）和内核代码（如路径解析中读目录）。

4.3 writei()——向 inode 写数据

int writei(struct inode *ip, int user_src, uint64 src, uint off, uint n)
{
  uint tot, m;
  struct buf *bp;

  if(off > ip->size || off + n < off)
    return -1;
  if(off + n > MAXFILE*BSIZE)
    return -1;   // 超过最大文件大小（268 KB）

  for(tot=0; tot<n; tot+=m, off+=m, src+=m) {
    bp = bread(ip->dev, bmap(ip, off/BSIZE));   // 读/分配目标块
    m  = min(n - tot, BSIZE - off%BSIZE);
    if(either_copyin(bp->data + (off % BSIZE), user_src, src, m) == -1) {
      brelse(bp);
      break;
    }
    log_write(bp);   // ← 关键：写入日志（不直接 bwrite）
    brelse(bp);
  }

  if(n > 0) {
    if(off > ip->size)
      ip->size = off;    // 文件扩大，更新 size
    iupdate(ip);         // 将 size（及可能新增的 addrs[]）写入日志
  }

  return n;  // 实际写入字节数（可能 < n，若 either_copyin 失败）
}

writei() 与 readi() 的关键区别：

  • writei() 调用 log_write(bp) 将修改记录到日志；readi() 只读不写日志
  • writei() 可能调用 bmap() 分配新块（bmap() 内部调用 balloc()），因此必须在事务（begin_op/end_op）内调用
  • writei() 写完后调用 iupdate(ip) 同步 inode 元数据（size 字段）

4.4 itrunc()——截断文件，释放所有数据块

void itrunc(struct inode *ip)
{
  int i, j;
  struct buf *bp;
  uint *a;

  // ① 释放所有直接块
  for(i = 0; i < NDIRECT; i++) {
    if(ip->addrs[i]) {
      bfree(ip->dev, ip->addrs[i]);
      ip->addrs[i] = 0;
    }
  }

  // ② 释放一级间接块
  if(ip->addrs[NDIRECT]) {
    bp = bread(ip->dev, ip->addrs[NDIRECT]);  // 读间接块
    a  = (uint*)bp->data;
    for(j = 0; j < NINDIRECT; j++) {
      if(a[j])
        bfree(ip->dev, a[j]);    // 释放每个数据块
    }
    brelse(bp);
    bfree(ip->dev, ip->addrs[NDIRECT]);  // 释放间接块本身
    ip->addrs[NDIRECT] = 0;
  }

  ip->size = 0;
  iupdate(ip);   // 将清零后的 addrs[] 和 size = 0 写回磁盘
}

itrunc() 在 iput() 中当 nlink == 0 时被调用，也在 sys_open() 以 O_TRUNC 打开文件时调用。

五、源码精读：目录操作

5.1 dirlookup()——在目录中查找名字

struct inode* dirlookup(struct inode *dp, char *name, uint *poff)
{
  uint off, inum;
  struct dirent de;

  if(dp->type != T_DIR)
    panic("dirlookup not DIR");

  // 目录内容是连续的 struct dirent 条目
  // struct dirent { ushort inum; char name[DIRSIZ]; }（DIRSIZ=14）
  for(off = 0; off < dp->size; off += sizeof(de)) {
    if(readi(dp, 0, (uint64)&de, off, sizeof(de)) != sizeof(de))
      panic("dirlookup read");
    if(de.inum == 0)
      continue;               // inum == 0 表示该条目已删除（空洞）
    if(namecmp(name, de.name) == 0) {
      if(poff)
        *poff = off;          // 返回条目的字节偏移（供 unlink 使用）
      inum = de.inum;
      return iget(dp->dev, inum);  // 返回对应 inode（未锁定，ref++）
    }
  }
  return 0;  // 未找到
}

目录本质上是一个文件，其数据内容是 struct dirent 的紧密排列：

目录数据块内容（字节偏移）：
  [0]   : dirent {inum=1, name="."}      （本目录自身）
  [16]  : dirent {inum=1, name=".."}     （父目录）
  [32]  : dirent {inum=5, name="README"} （文件条目）
  [48]  : dirent {inum=0, name=""}       （被删除的条目，inum=0 跳过）
  [64]  : dirent {inum=7, name="ls"}     （另一个文件）
  ...

struct dirent 大小 = sizeof(ushort) + DIRSIZ = 2 + 14 = 16 字节。

5.2 dirlink()——在目录中添加新条目

int dirlink(struct inode *dp, char *name, uint inum)
{
  int off;
  struct dirent de;
  struct inode *ip;

  // ① 检查名字是否已存在（重复则失败）
  if((ip = dirlookup(dp, name, 0)) != 0) {
    iput(ip);
    return -1;
  }

  // ② 找第一个空槽（inum == 0 的条目）或目录末尾
  for(off = 0; off < dp->size; off += sizeof(de)) {
    if(readi(dp, 0, (uint64)&de, off, sizeof(de)) != sizeof(de))
      panic("dirlink read");
    if(de.inum == 0)
      break;   // 找到空槽
  }
  // off == dp->size 表示没有空槽，在末尾追加（writei 会自动扩大 dp->size）

  // ③ 写入新条目
  strncpy(de.name, name, DIRSIZ);
  de.inum = inum;
  if(writei(dp, 0, (uint64)&de, off, sizeof(de)) != sizeof(de))
    panic("dirlink");

  return 0;
}

dirlink() 复用删除条目的空槽：unlink() 只将条目的 inum 清零（软删除），dirlink() 会优先填充这些空槽，避免目录无限增长。

六、源码精读：路径解析

6.1 skipelem()——从路径中提取一个组件

static char* skipelem(char *path, char *name)
{
  char *s;
  int len;

  while(*path == '/')   // 跳过前导斜杠
    path++;
  if(*path == 0)        // 路径结束
    return 0;

  s = path;
  while(*path != '/' && *path != 0)  // 找到下一个 / 或结尾
    path++;
  len = path - s;                     // 当前组件的长度

  if(len >= DIRSIZ)
    memmove(name, s, DIRSIZ);         // 截断到 DIRSIZ（14 字节）
  else {
    memmove(name, s, len);
    name[len] = 0;                    // 添加终止符
  }

  while(*path == '/')   // 跳过组件后的多余斜杠
    path++;

  return path;  // 返回剩余路径（调用者用于下次迭代）
}

示例：

skipelem("a/bb/c",  name) → name="a",  返回 "bb/c"
skipelem("///a//bb", name) → name="a",  返回 "bb"
skipelem("a",       name) → name="a",  返回 ""
skipelem("",        name) → 返回 0（无更多组件）

6.2 namex()——核心路径解析函数

static struct inode* namex(char *path, int nameiparent, char *name)
{
  struct inode *ip, *next;

  // ① 确定起始 inode
  if(*path == '/')
    ip = iget(ROOTDEV, ROOTINO);  // 绝对路径：从根目录（inum=1）开始
  else
    ip = idup(myproc()->cwd);     // 相对路径：从当前工作目录开始

  // ② 逐组件遍历
  while((path = skipelem(path, name)) != 0) {
    ilock(ip);                    // 锁定当前目录（读 type 和 size 需要锁）

    if(ip->type != T_DIR) {       // 中间组件必须是目录
      iunlockput(ip);
      return 0;
    }

    if(nameiparent && *path == '\0') {
      // 调用方要求返回父目录，且 name 已是最后一个组件
      iunlock(ip);               // 解锁但保持引用（返回给调用者）
      return ip;
    }

    if((next = dirlookup(ip, name, 0)) == 0) {
      iunlockput(ip);            // 名字不存在
      return 0;
    }
    iunlockput(ip);              // 释放当前目录（找到下一层了）
    ip = next;                   // 移动到下一层 inode
  }

  if(nameiparent) {
    iput(ip);
    return 0;   // nameiparent 要求返回父目录，但路径无父（如 nameiparent "/"）
  }
  return ip;    // 返回最终 inode（未锁定，ref > 0）
}

namex() 的并发陷阱——为何 ilock 和 iget 分开？

考虑路径 /a/b，遍历第一步：

  1. ilock(root_dir) — 锁定根目录
  2. dirlookup(root_dir, "a", 0) — 在根目录里查找 “a”，返回 next = iget(dev, a_inum)（ref++，未锁）
  3. iunlockput(root_dir) — 释放根目录锁，同时减少 ref
  4. ip = next（即 /a 的 inode）

步骤 3 之前，另一个进程可能删除 /a（nlink → 0），但因为步骤 2 已经 iget 增加了 ref，iput() 不会立即释放，确保指针有效。这就是 分离引用（ref）和锁定（lock） 的意义：在不持锁时也能安全持有指针。

6.3 namei() 和 nameiparent()

struct inode* namei(char *path)
{
  char name[DIRSIZ];
  return namex(path, 0, name);
  // nameiparent=0：返回路径最终的 inode
}

struct inode* nameiparent(char *path, char *name)
{
  return namex(path, 1, name);
  // nameiparent=1：返回最后一个组件的父目录，并将最后一个组件名写入 name
}

使用场景：

函数 用途 示例
namei("/etc/passwd") 获取文件的 inode sys_open("/etc/passwd", ...)
nameiparent("/etc/passwd", name) 获取父目录 /etc，name="passwd" sys_create、sys_unlink

七、inode 操作的完整时序：以创建文件为例

sys_open("/test", O_CREATE | O_WRONLY)
  │
  ├── begin_op()
  │
  ├── nameiparent("/test", name)   name="test"
  │     └── namex: iget(root) → ilock(root) → 解析完成，返回 root（未锁）
  │
  ├── ilock(dp)   （锁定父目录 = 根目录）
  │
  ├── dirlookup(dp, "test", 0) → 返回 0（"test" 不存在）
  │
  ├── ialloc(dp->dev, T_FILE)
  │     └── 扫描 dinode，找空闲 inum，置 type=T_FILE，log_write
  │     └── iget(dev, inum) → 返回 ip（ref=1，valid=0，未锁）
  │
  ├── ilock(ip)
  │     └── valid=0 → bread inode块，读 dinode → valid=1
  │
  ├── ip->major=0, ip->minor=0, ip->nlink=1
  ├── iupdate(ip)              → 将 nlink=1 写入日志
  │
  ├── dirlink(dp, "test", ip->inum)
  │     └── 找空槽，writei 写 {inum, "test"} 到目录数据块 → log_write
  │
  ├── iunlock(dp)
  │
  └── end_op()                 → 最后一个操作触发 commit()

九、本节小结

概念 要点
struct dinode 磁盘 inode：type/nlink/size/addrs[13]（12 直接 + 1 间接）
struct inode 内存 inode：磁盘字段 + ref/valid/lock（sleeplock）
balloc/bfree 位图管理数据块；balloc 按需清零新块
ialloc 扫描 dinode 找 type=0，设置类型，log_write
iget 查找或创建缓存条目，仅增加引用计数，不加锁不读磁盘
idup 已有指针时仅 ref++
ilock 获取 sleeplock；若 valid=0 则从磁盘读 dinode
iput 减少 ref；nlink==0 && ref→0 时截断并释放磁盘 inode
iupdate 将内存 inode 字段同步回磁盘（write-through，经日志）
bmap 逻辑块号→物理块号，按需分配（balloc）
readi/writei 循环分块读写，跨块透明，writei 经 log_write
itrunc 释放所有直接块 + 间接块 + 间接块本身
dirlookup 线性扫描 dirent，按名字匹配，返回 iget 的 inode
dirlink 找空槽或追加，writei 写入 {inum, name}
skipelem 从路径提取一个组件，返回剩余路径
namex 循环 skipelem + ilock + dirlookup，逐层解析路径
ref vs nlink ref：内存引用（icache.lock 保护）；nlink：磁盘链接数（ip->lock 保护）

使用 inode 的正确模式：

// 打开/查找文件：
ip = namei(path);          // 或 ialloc()
ilock(ip);                 // 锁定（此时读取 valid 字段并按需读磁盘）
// ... 读写 ip->size, ip->addrs, readi(), writei() ...
iunlock(ip);               // 解锁（不释放引用，如文件描述符长期持有）

// 完全释放：
iput(ip);                  // 减少 ref，nlink=0 时释放磁盘资源

下一节预告：第 16 节（文件描述符与系统调用层）将在 inode 层之上构建 struct file、全局文件表和系统调用接口（sys_open/read/write/close），把 inode、管道、设备统一成”一切皆文件”的抽象。
